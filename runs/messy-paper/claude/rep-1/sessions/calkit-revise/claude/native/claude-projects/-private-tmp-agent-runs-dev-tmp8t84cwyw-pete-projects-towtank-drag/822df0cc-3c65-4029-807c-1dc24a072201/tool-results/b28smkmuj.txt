=== scripts/compute_cd.py
"""Compute drag coefficient, Reynolds number, and related parameters per run.

Writes a per-run table and a JSON summary whose keys are referenced from the
paper via the json-to-latex stage.
"""

import json

import numpy as np
import pandas as pd

from towtank import G, POST_WINDOW, STEADY_WINDOW, TARE_WINDOW

L_SUB = 0.60  # Submerged cylinder length (m)
TANK_WIDTH = 0.90  # m
TANK_DEPTH = 0.75  # m
# Runs whose load cell zero moved by more than this over the run are excluded
MAX_ZERO_DRIFT_N = 0.05

# Labels used in the run log, mapped to the key stems used in the paper
CYLINDERS = {"1in": "OneIn", "1.5in": "OnePointFiveIn", "2in": "TwoIn"}


def water_density(temp_c):
    """Density of air-free pure water (kg/m^3), Tanaka et al. (2001)."""
    a1, a2, a3, a4, a5 = -3.983035, 301.797, 522528.9, 69.34881, 999.974950
    t = temp_c
    return a5 * (1 - (t + a1) ** 2 * (t + a2) / (a3 * (t + a4)))


def water_viscosity(temp_c):
    """Dynamic viscosity of water (Pa s) from the Vogel equation."""
    return 2.414e-5 * 10 ** (247.8 / (temp_c + 273.15 - 140.0))


forces = pd.read_csv("results/forces.csv")
log = pd.read_csv("data/run_log.csv")
df = log.merge(forces, left_on="run_id", right_on="run")

d = df.diameter_mm / 1000
u = df.speed_m_s
df["rho"] = water_density(df.water_temp_c)
df["nu"] = water_viscosity(df.water_temp_c) / df.rho
df["Re"] = u * d / df.nu
df["Fr_D"] = u / np.sqrt(G * d)
df["Cd"] = df.F_mean_N / (0.5 * df.rho * u**2 * d * L_SUB)
# Cd with the fluctuating force std, to show the unsteadiness from shedding
df["Cd_std"] = df.F_std_N / (0.5 * df.rho * u**2 * d * L_SUB)
df["excluded"] = df.zero_drift_N.abs() > MAX_ZERO_DRIFT_N

cols = [
    "run_id", "date", "cylinder", "diameter_mm", "speed_m_s",
    "water_temp_c", "rho", "nu", "Re", "Fr_D", "F_mean_N", "F_std_N",
    "zero_drift_N", "Cd", "Cd_std", "excluded",
]
df[cols].to_csv("results/cd.csv", index=False)

good = df[~df.excluded]
excluded = df[df.excluded]
summary = {
    "NRuns": int(len(df)),
    "NRunsUsed": int(len(good)),
    "NRunsExcluded": int(len(excluded)),
    "ExcludedRuns": ", ".join(str(r) for r in excluded.run_id),
    "ExcludedZeroDrift": ", ".join(
        f"{x:.1f}" for x in excluded.zero_drift_N
    ),
    "MaxZeroDriftThreshold": MAX_ZERO_DRIFT_N,
    "MaxZeroDriftUsed": round(good.zero_drift_N.abs().max(), 3),
    "TareStart": TARE_WINDOW[0],
    "TareEnd": TARE_WINDOW[1],
    "SteadyStart": STEADY_WINDOW[0],
    "SteadyEnd": STEADY_WINDOW[1],
    "PostStart": POST_WINDOW[0],
    "PostEnd": POST_WINDOW[1],
    "SubmergedLength": L_SUB,
    "TankWidth": TANK_WIDTH,
    "TankDepth": TANK_DEPTH,
    "SpeedMin": float(df.speed_m_s.min()),
    "SpeedMax": float(df.speed_m_s.max()),
    "NSpeeds": int(df.speed_m_s.nunique()),
    "TempMin": float(df.water_temp_c.min()),
    "TempMax": float(df.water_temp_c.max()),
    "ReMin": int(round(good.Re.min(), -2)),
    "ReMax": int(round(good.Re.max(), -2)),
    "FrMin": round(good.Fr_D.min(), 2),
    "FrMax": round(good.Fr_D.max(), 2),
    "SensOld": round(df[df.date < "2026-03-10"].sens_V_per_N.iloc[0], 5),
    "SensNew": round(df[df.date >= "2026-03-10"].sens_V_per_N.iloc[0], 5),
    "CalResidMax": round(df.cal_resid_max_N.max(), 3),
    "CdMeanAll": round(good.Cd.mean(), 3),
    "CdStdAll": round(good.Cd.std(), 3),
    "CdMinAll": round(good.Cd.min(), 3),
    "CdMaxAll": round(good.Cd.max(), 3),
    "CdExcluded": ", ".join(f"{x:.2f}" for x in excluded.Cd),
    "CdFluctMax": round(good.Cd_std.max(), 2),
}
for label, stem in CYLINDERS.items():
    g = df[df.cylinder == label]
    gg = good[good.cylinder == label]
    dia = g.diameter_mm.iloc[0] / 1000
    summary[f"D{stem}"] = float(g.diameter_mm.iloc[0])
    summary[f"Date{stem}"] = g.date.iloc[0]
    summary[f"Temp{stem}"] = round(g.water_temp_c.mean(), 1)
    summary[f"N{stem}"] = int(len(gg))
    summary[f"CdMean{stem}"] = round(gg.Cd.mean(), 3)
    summary[f"CdStd{stem}"] = round(gg.Cd.std(), 3)
    summary[f"ReMin{stem}"] = int(round(gg.Re.min(), -2))
    summary[f"ReMax{stem}"] = int(round(gg.Re.max(), -2))
    summary[f"AspectRatio{stem}"] = round(L_SUB / dia, 1)
    # Width blockage D/W and frontal-area blockage D L / (W H), in percent
    summary[f"BlockageWidth{stem}"] = round(100 * dia / TANK_WIDTH, 1)
    summary[f"BlockageArea{stem}"] = round(
        100 * dia * L_SUB / (TANK_WIDTH * TANK_DEPTH), 1
    )

with open("results/summary.json", "w") as f:
    json.dump(summary, f, indent=2)

=== scripts/process_runs.py
"""Reduce every raw run to a mean drag force.

Replaces process.py and process_b3.py: one code path for both DAQ formats,
with the sensitivity fitted from the calibration valid for each batch.
"""

import pandas as pd

from towtank import (
    BATCHES,
    POST_WINDOW,
    STEADY_WINDOW,
    TARE_WINDOW,
    calibration,
    read_run,
    window,
)

log = pd.read_csv("data/run_log.csv")

rows = []
for _, r in log.iterrows():
    batch = r.date
    if batch not in BATCHES:
        raise ValueError(f"No batch config for {batch}")
    sens, cal_resid_N = calibration(batch)
    t, v = read_run(batch, r.run_id)
    tare = v[window(t, TARE_WINDOW)].mean()
    post = v[window(t, POST_WINDOW)].mean()
    force = (v - tare) / sens
    steady = force[window(t, STEADY_WINDOW)]
    rows.append(
        {
            "run": r.run_id,
            "sens_V_per_N": sens,
            "cal_resid_max_N": cal_resid_N,
            "F_mean_N": steady.mean(),
            "F_std_N": steady.std(),
            # Shift of the zero over the run; a large value means the
            # tare can't be trusted
            "zero_drift_N": (post - tare) / sens,
        }
    )

pd.DataFrame(rows).to_csv("results/forces.csv", index=False)
=== scripts/towtank.py
"""Shared helpers for reading tow tank runs and converting them to force."""

import os

import numpy as np
import pandas as pd

# Each test day, with its DAQ file format and the load cell calibration that
# was valid at the time (the load cell was remounted and recalibrated on
# 2026-03-10, see notes.md)
BATCHES = {
    "2025-10-14": {"fmt": "v2.1", "cal": "loadcell_2025-09.csv"},
    "2025-11-03": {"fmt": "v2.1", "cal": "loadcell_2025-09.csv"},
    "2026-03-12": {"fmt": "semicolon", "cal": "loadcell_2026-03.csv"},
}
RAW_DIR = "data/raw"
CAL_DIR = "data/calibration"
G = 9.81

# Time windows (s): load cell zero before the carriage starts, steady towing,
# and the zero after the carriage has stopped
TARE_WINDOW = (0.0, 1.0)
STEADY_WINDOW = (3.0, 8.0)
POST_WINDOW = (11.0, 12.0)


def run_path(batch, run):
    return os.path.join(RAW_DIR, batch, f"run_{run:03d}.csv")


def read_run(batch, run):
    """Return time (s) and load cell voltage (V) for one run."""
    fmt = BATCHES[batch]["fmt"]
    path = run_path(batch, run)
    if fmt == "v2.1":
        d = pd.read_csv(path, comment="#")
        return d.time_s.values, d.loadcell_V.values
    elif fmt == "semicolon":
        d = pd.read_csv(path, sep=";")
        return d.t.values, d.V.values
    raise ValueError(f"Unknown DAQ format {fmt}")


def calibration(batch):
    """Fit V = sens * F + offset to the static calibration for a batch.

    Returns sensitivity (V/N) and the maximum fit residual expressed in N.
    """
    cal = pd.read_csv(os.path.join(CAL_DIR, BATCHES[batch]["cal"]))
    force = cal.mass_g / 1000 * G
    sens, offset = np.polyfit(force, cal.voltage_V, 1)
    resid = cal.voltage_V - (sens * force + offset)
    return sens, np.abs(resid).max() / sens


def window(t, lo_hi):
    return (t >= lo_hi[0]) & (t < lo_hi[1])

\begin{abstract}
We report mean drag measurements on three smooth circular cylinders, of
diameter \result[DOneIn], \result[DOnePointFiveIn], and
\result[DTwoIn]~\si{mm}, towed vertically through a water tank at

\section{Introduction}
\label{sec:intro}

The flow past a circular cylinder is one of the canonical problems of
bluff-body aerodynamics and hydrodynamics. Over the subcritical range of
Reynolds number, roughly $10^3 \lesssim \Rey \lesssim 2\times10^5$, the
boundary layer separates laminar, the wake sheds vortices periodically, and
the drag coefficient of a nominally two-dimensional cylinder is nearly
constant~\citep{Wieselsberger1921,Roshko1961,Williamson1996,Zdravkovich1997}.
Measured values in practice also depend on blockage, aspect ratio, and end
conditions~\citep{WestApelt1982,Zdravkovich1997}.
\todo[inline]{Motivation: why these measurements? E.g., facility
commissioning, validation data for a course or for simulations, a

This paper describes tow tank measurements of the mean drag on three smooth
cylinders over \result[ReMin]~$\le \Rey \le$~\result[ReMax], the data
reduction used to obtain drag coefficients, and how the results compare with
the classical data.

\section{Data reduction}
\label{sec:reduction}

Figure~\ref{fig:timeseries}(top) shows a typical force record. The carriage
was at rest for the first part of each record, accelerated to a constant
speed, towed, and decelerated back to rest. The load cell zero was taken as
the mean voltage over $\result[TareStart] \le t < \result[TareEnd]~\si{s}$,
before the carriage started moving. The mean drag $\bar{F}$ is the mean force
over the steady towing window
$\result[SteadyStart] \le t < \result[SteadyEnd]~\si{s}$. The same window was
used for every run.
\todo[inline]{Justify the window, e.g., by the number of shedding cycles it
contains and how much the mean changes when it is shortened.}
The force fluctuations visible in the steady window are due to vortex
shedding.

The drag coefficient and Reynolds number are
\begin{equation}
  C_D = \frac{\bar{F}}{\tfrac{1}{2}\rho U^2 D L}, \qquad
  \Rey = \frac{U D}{\nu},
\end{equation}
where $U$ is the carriage speed. The water density $\rho$ and kinematic
viscosity $\nu$ were evaluated at each run's measured temperature. Density
was computed from the formula of \citet{Tanaka2001} and dynamic viscosity
from the Vogel equation $\mu = A\,10^{B/(T-C)}$ with
$A = 2.414\times10^{-5}~\si{Pa.s}$, $B = 247.8~\si{K}$, and
$C = 140~\si{K}$.
\todo[inline]{Cite a source for these Vogel constants, or switch to a standard
reference correlation such as IAPWS.}
The diameter-based Froude number $\Fr = U/\sqrt{gD}$ ranged from
\result[FrMin] to \result[FrMax].

\paragraph{Run exclusion.}
The zero was also measured after each run, over
$\result[PostStart] \le t < \result[PostEnd]~\si{s}$. A run was excluded if
the zero moved by more than \result[MaxZeroDriftThreshold]~\si{N} over the
run. Only run \result[ExcludedRuns] met
this criterion: its zero shifted by \result[ExcludedZeroDrift]~\si{N}, and
its force rose steadily through the towing window
(Fig.~\ref{fig:timeseries}, bottom). The run was flagged in the run log at
the time of testing. For all retained runs the zero moved by at most
\result[MaxZeroDriftUsed]~\si{N}.

\section{Results}
\label{sec:results}

Figure~\ref{fig:cd} shows $C_D$ against $\Rey$ for all runs, and
Table~\ref{tab:summary} summarizes each cylinder. Across the
\result[NRunsUsed] retained runs, $C_D$ ranges from \result[CdMinAll] to
\result[CdMaxAll], with a mean of \result[CdMeanAll] and standard deviation
\result[CdStdAll]. The per-cylinder means, \result[CdMeanOneIn] for
$D = \result[DOneIn]~\si{mm}$, \result[CdMeanOnePointFiveIn] for
$D = \result[DOnePointFiveIn]~\si{mm}$, and \result[CdMeanTwoIn] for

\section{Discussion}
\label{sec:discussion}

\paragraph{Comparison with classical data.}
The measured drag coefficients fall within the range usually quoted for
circular cylinders in the subcritical regime~\citep{Wieselsberger1921,Zdravkovich1997}.
\todo[inline]{Overlay or tabulate reference data (e.g., Wieselsberger, or the
compilation in Zdravkovich) on Fig.~\ref{fig:cd} and compare quantitatively.
Any reference data should enter through the pipeline as an imported dataset,
not be typed in here.}

\paragraph{Blockage.}
The blockage ratios in Table~\ref{tab:summary} are small but not negligible,
and they double from the smallest cylinder to the largest. Blockage raises
the measured drag, so a correction~\citep[e.g.,][]{Maskell1963,WestApelt1982}
would lower $C_D$, more for the larger cylinders. The near-equality of the
uncorrected per-cylinder means therefore does not carry over automatically
to the corrected values.
\todo[inline]{Decide on a blockage correction method (the geometry is a
surface-piercing cylinder spanning part of the depth, so the choice is not
obvious), add it as a pipeline stage, and report corrected $C_D$.}

\paragraph{Aspect ratio, free surface, and end effects.}
The submerged aspect ratio ranges from \result[AspectRatioTwoIn] to
\result[AspectRatioOneIn]. A finite span with a free lower end generally
reduces the sectional drag relative to a two-dimensional
cylinder~\citep{Zdravkovich1997}. At the upper end the cylinder pierces the
free surface, and with $\Fr$ up to \result[FrMax] the surface deformation
and wave making at the higher speeds may contribute to the measured force.
\todo[inline]{Discuss these effects against the literature, and say whether
any surface deformation was observed during the runs.}

\paragraph{Uncertainty.}
\todo[inline]{Uncertainty analysis: calibration (sensitivity fit, at most
\result[CalResidMax]~\si{N} residual), zero, the statistical uncertainty of
the mean over the window (account for the correlation from shedding), speed,
diameter, submerged length, and temperature. Propagate to $C_D$ and $\Rey$,
add error bars to Fig.~\ref{fig:cd}, and use the result to judge whether the
diameter and Reynolds number differences in Sect.~\ref{sec:results} are
significant.}

\section{Conclusions}
\label{sec:conclusions}

Mean drag was measured on three smooth circular cylinders towed through a
water tank over \result[ReMin]~$\le \Rey \le$~\result[ReMax]. With fluid
properties evaluated at the measured water temperature and one run excluded
for zero drift, the drag coefficient averaged \result[CdMeanAll]. The per-cylinder means agree within
their run-to-run scatter, before any blockage correction or uncertainty
analysis.

literature comparison are in.}
