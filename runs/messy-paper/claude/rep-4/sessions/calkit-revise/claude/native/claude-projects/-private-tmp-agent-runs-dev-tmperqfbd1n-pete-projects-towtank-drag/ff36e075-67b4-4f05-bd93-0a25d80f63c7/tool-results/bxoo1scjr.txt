# towtank-drag

Drag on smooth circular cylinders towed through the tank. 1in, 2in and
1.5in cylinders, 0.6 m submerged.

This is a [Calkit](https://calkit.org) project. Running

```sh
calkit run
```

runs the whole pipeline (defined in `calkit.yaml`), from the raw data to the
paper, re-running only stages whose inputs have changed:

1. `process` (`scripts/process.py`): raw load cell records -> mean force per
   run (`results/forces.csv`), with the calibration fitted from the file in
   effect for each batch. Reads both DAQ file formats.
2. `analyze` (`scripts/analyze.py`): Cd, Re (water properties at the
   logged temperature), Fr, St, and run screening (load cell zero drift).
   Writes `results/cd.csv`, `results/summary.json`, and the numbers and
   tables used in the paper (`paper/generated/`).
3. `plot-figures` (`scripts/plot_figures.py`): figures in `figures/paper/`.
4. `build-paper`: compiles `paper/paper.tex` -> `paper/paper.pdf` in
   Docker (TeX Live), so no local LaTeX install is needed.

The Python environment (`py`, from `requirements.txt`) is created
automatically with uv. Raw data in `data/raw` is tracked with DVC.

Calibration files are in data/calibration. Run log is data/run_log.csv.

## Legacy files

`process.py`, `process_b3.py`, `plot_timeseries.py`, `analysis.ipynb`,
`processed/` and the existing `figures/*.png` are the original workflow,
replaced by the pipeline above. old_matlab/ is Jamie's original
processing script, kept for reference.

"""Compute drag coefficient, Reynolds, Froude, and Strouhal numbers per run.

Water properties are evaluated at the logged water temperature of each run.
Runs whose post-run zero differs from the pre-run tare by more than
DRIFT_LIMIT_N are flagged and excluded from the summary statistics.

Outputs results/cd.csv, results/summary.json, and LaTeX snippets in
paper/generated/ for the manuscript.
"""

import json
import os

import numpy as np
import pandas as pd

OUT_DIR = "results"
TEX_DIR = "paper/generated"

G = 9.81
L = 0.60  # m, submerged length of cylinder
TANK_WIDTH = 0.90  # m
TANK_DEPTH = 0.75  # m
DRIFT_LIMIT_N = 0.05
MASKELL_EPS = 2.5  # Maskell (1963) blockage factor, for an indicative correction


def water_viscosity(T):
    """Dynamic viscosity (Pa s), Kestin et al. (1978) correlation."""
    dT = 20.0 - T
    log_ratio = dT / (T + 96) * (1.2378 - 1.303e-3 * dT + 3.06e-6 * dT**2
                                 + 2.55e-8 * dT**3)
    return 1.002e-3 * 10**log_ratio


def water_density(T):
    """Density (kg/m^3), Tanaka et al. (2001) formula for air-free water."""
    a1, a2, a3, a4, a5 = -3.983035, 301.797, 522528.9, 69.34881, 999.974950
    return a5 * (1 - (T + a1) ** 2 * (T + a2) / (a3 * (T + a4)))


forces = pd.read_csv(os.path.join(OUT_DIR, "forces.csv"))
log = pd.read_csv("data/run_log.csv").rename(columns={"run_id": "run"})
df = forces.merge(log, on="run", validate="one_to_one")

D = df.diameter_mm / 1000
U = df.speed_m_s
df["rho"] = water_density(df.water_temp_c)
df["nu"] = water_viscosity(df.water_temp_c) / df.rho
df["Re"] = U * D / df.nu
df["Fr_D"] = U / np.sqrt(G * D)
q_area = 0.5 * df.rho * U**2 * D * L
df["Cd"] = df.F_mean_N / q_area
df["Cd_sem"] = df.F_sem_N / q_area
df["Cd_rms"] = df.F_std_N / q_area
df["St"] = df.f_peak_Hz / 2 * D / U  # drag fluctuates at twice the shedding freq.
df["blockage"] = D * L / (TANK_WIDTH * TANK_DEPTH)
df["Cd_maskell"] = df.Cd / (1 + MASKELL_EPS * df.Cd * df.blockage)
df["excluded"] = df.zero_drift_N.abs() > DRIFT_LIMIT_N
df.to_csv(os.path.join(OUT_DIR, "cd.csv"), index=False)

good = df[~df.excluded]
cyl_order = ["1in", "1.5in", "2in"]
by_cyl = {}
for cyl in cyl_order:
    g = good[good.cylinder == cyl]
    by_cyl[cyl] = {
        "D_mm": float(g.diameter_mm.iloc[0]),
        "aspect_ratio": float(L / (g.diameter_mm.iloc[0] / 1000)),
        "blockage": float(g.blockage.iloc[0]),
        "n_runs": int(len(g)),
        "Cd_mean": float(g.Cd.mean()),
        "Cd_std": float(g.Cd.std()),
        "Cd_maskell_mean": float(g.Cd_maskell.mean()),
        "Re_min": float(g.Re.min()),
        "Re_max": float(g.Re.max()),
        "T_min": float(g.water_temp_c.min()),
        "T_max": float(g.water_temp_c.max()),
    }

# Re that would have been reported with the old fixed 20 C viscosity
re_20c_bias = float((U * D / 1.004e-6 / df.Re - 1).loc[~df.excluded].mean())
excluded = df[df.excluded]
with open(os.path.join(OUT_DIR, "calibration.json")) as f:
    cals = json.load(f)
rel_sens_err = {
    f: s["sensitivity_std_err"] / s["sensitivity_V_per_N"] for f, s in cals.items()
}
cd_vs_fr = np.polyfit(good.Fr_D, good.Cd, 1)
summary = {
    "n_runs": int(len(df)),
    "n_good": int(len(good)),
    "excluded_runs": excluded.run.tolist(),
    "excluded_drift_N": excluded.zero_drift_N.round(3).tolist(),
    "max_drift_good_N": float(good.zero_drift_N.abs().max()),
    "Cd_mean_all": float(good.Cd.mean()),
    "Cd_std_all": float(good.Cd.std()),
    "Cd_maskell_mean_all": float(good.Cd_maskell.mean()),
    "Cd_excluded_run": excluded.Cd.round(3).tolist(),
    "Cd_sem_max_rel": float((good.Cd_sem / good.Cd).max()),
    "Cd_rms_mean_rel": float((good.Cd_rms / good.Cd).mean()),
    "Re_min": float(good.Re.min()),
    "Re_max": float(good.Re.max()),
    "Fr_min": float(good.Fr_D.min()),
    "Fr_max": float(good.Fr_D.max()),
    "Cd_vs_Fr_slope": float(cd_vs_fr[0]),
    "Cd_vs_Fr_corr": float(np.corrcoef(good.Fr_D, good.Cd)[0, 1]),
    "Cd_vs_Re_corr": float(np.corrcoef(np.log(good.Re), good.Cd)[0, 1]),
    "St_mean": float(good.St.mean()),
    "St_std": float(good.St.std()),
    "re_20c_bias": re_20c_bias,
    "sensitivity_rel_err_max": float(max(rel_sens_err.values())),
    "by_cylinder": by_cyl,
}
with open(os.path.join(OUT_DIR, "summary.json"), "w") as f:
    json.dump(summary, f, indent=2)

# LaTeX macros so numbers in the text always match the pipeline output
os.makedirs(TEX_DIR, exist_ok=True)
names = {"1in": "One", "1.5in": "OneHalf", "2in": "Two"}

    "CalResidMax": f"{max(c['max_resid_N'] for c in cals.values()) * 1000:.0f}",
    "CalMaxLoad": f"{max(c['max_load_N'] for c in cals.values()):.1f}",
    "FMaxGood": f"{good.F_mean_N.max():.1f}",
    "SubLength": f"{L:.2f}",
    "EndGap": f"{TANK_DEPTH - L:.2f}",
    "TankWidth": f"{TANK_WIDTH:.2f}",
    "TankDepth": f"{TANK_DEPTH:.2f}",
    "MaskellEps": f"{MASKELL_EPS:.1f}",
}
for cyl, s in by_cyl.items():
    n = names[cyl]
    macros[f"Cd{n}"] = f"{s['Cd_mean']:.3f}"
    macros[f"CdStd{n}"] = f"{s['Cd_std']:.3f}"

        f.write(f"\\newcommand{{\\{k}}}{{{v}}}\n")

# Summary table of the three cylinders

"""Reduce raw tow tank load cell time series to mean drag force per run.

Handles both DAQ file formats (comma-separated with a commented header for
the 2025 batches, semicolon-separated for the 2026 batch) and fits the load
cell sensitivity from the calibration file in effect on the run date.

Outputs results/forces.csv and results/calibration.json.
"""

import glob
import json
import os

import numpy as np
import pandas as pd

RAW_DIR = "data/raw"
CAL_DIR = "data/calibration"
OUT_DIR = "results"

G = 9.81
FS = 200.0  # Hz, sample rate for both DAQ versions
TARE_WINDOW = (0.0, 1.0)  # s, carriage at rest before the run
STEADY_WINDOW = (3.0, 8.0)  # s, constant carriage speed
POST_WINDOW = (11.0, 12.0)  # s, carriage at rest after the run
N_BLOCKS = 5  # blocks for standard error of the steady-state mean

# Calibration in effect for each batch (load cell remounted 2026-03-10)
CALIBRATIONS = {
    "2025-10-14": "loadcell_2025-09.csv",
    "2025-11-03": "loadcell_2025-09.csv",
    "2026-03-12": "loadcell_2026-03.csv",
}


def fit_calibration(fname):
    cal = pd.read_csv(os.path.join(CAL_DIR, fname))
    force = cal.mass_g / 1000 * G
    (slope, intercept), cov = np.polyfit(force, cal.voltage_V, 1, cov=True)
    resid = cal.voltage_V - (slope * force + intercept)
    return {
        "file": fname,
        "sensitivity_V_per_N": float(slope),
        "sensitivity_std_err": float(np.sqrt(cov[0, 0])),
        "offset_V": float(intercept),
        "max_resid_N": float(np.abs(resid).max() / slope),
        "max_load_N": float(force.max()),
        "n_points": len(cal),
    }


def read_run(fname):
    """Return (time_s, voltage_V) for either DAQ file format."""
    with open(fname) as f:
        first = f.readline()
    if ";" in first:
        d = pd.read_csv(fname, sep=";")
        return d.t.values, d.V.values
    d = pd.read_csv(fname, comment="#")
    return d.time_s.values, d.loadcell_V.values


def in_window(t, window):
    return (t >= window[0]) & (t < window[1])


def dominant_frequency(x):
    n = np.arange(len(x))
    x = x - np.polyval(np.polyfit(n, x, 1), n)
    power = np.abs(np.fft.rfft(x * np.hanning(len(x)))) ** 2
    freqs = np.fft.rfftfreq(len(x), 1 / FS)
    return freqs[np.argmax(power[1:]) + 1]


cals = {name: fit_calibration(name) for name in sorted(set(CALIBRATIONS.values()))}
for c in cals.values():
    print(f"{c['file']}: sensitivity = {c['sensitivity_V_per_N']:.5f} V/N")

rows = []
for batch, cal_file in CALIBRATIONS.items():
    sens = cals[cal_file]["sensitivity_V_per_N"]
    for fname in sorted(glob.glob(os.path.join(RAW_DIR, batch, "run_*.csv"))):
        t, v = read_run(fname)
        tare = v[in_window(t, TARE_WINDOW)].mean()
        post = v[in_window(t, POST_WINDOW)].mean()
        force = (v - tare) / sens
        steady = force[in_window(t, STEADY_WINDOW)]
        block_means = np.array([b.mean() for b in np.array_split(steady, N_BLOCKS)])
        rows.append({
            "run": int(os.path.basename(fname)[4:7]),
            "batch": batch,
            "calibration": cal_file,
            "sensitivity_V_per_N": sens,
            "F_mean_N": steady.mean(),
            "F_std_N": steady.std(),
            "F_sem_N": block_means.std(ddof=1) / np.sqrt(N_BLOCKS),
            "zero_drift_N": (post - tare) / sens,
            "f_peak_Hz": dominant_frequency(steady),
        })

os.makedirs(OUT_DIR, exist_ok=True)
out = pd.DataFrame(rows).sort_values("run")
out.to_csv(os.path.join(OUT_DIR, "forces.csv"), index=False)
with open(os.path.join(OUT_DIR, "calibration.json"), "w") as f:
    json.dump(cals, f, indent=2)
print(f"Processed {len(out)} runs")
